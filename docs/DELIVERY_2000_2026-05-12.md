## Final delivery package (Thu morning)

```bash
# Merge all pass JSONLs + dedup by query_id → final 2000
python3 -c "
import json, glob, pathlib

seen, out = set(), []
sources = [
    'data/03_queries/delivery_v1_2026-04-13_intra_doc.jsonl',
    'data/03_queries/m2_m15_reasoning_path_prod_20260511/m2_m15_reasoning_path_pass.jsonl',
    'data/03_queries/m2_diverse_v1_hub_kb_pass.jsonl',
    'data/03_queries/long_chain_iterative_pass.jsonl',
] + sorted(glob.glob('data/03_queries/delivery_sweep_2000/*_pass.jsonl'))

for fp in sources:
    if not pathlib.Path(fp).exists(): continue
    for line in open(fp):
        r = json.loads(line)
        qid = r.get('query_id') or r.get('pair_id')
        if qid and qid not in seen:
            seen.add(qid); out.append(r)
            if len(out) >= 2000: break
    if len(out) >= 2000: break

with open('data/03_queries/delivery_v2_20260514_FINAL.jsonl', 'w') as f:
    for r in out:
        f.write(json.dumps(r, ensure_ascii=False) + '\\n')

print(f'Final delivery: {len(out)} queries')
"
```
