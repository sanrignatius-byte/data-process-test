\item Problem setting:
  \begin{itemize}
    \item Given a set of instances $Z$, for each instance $\z = (\x, g, y)$ in $Z$, it has a k-dimensional feature vector $\x \in \R^k$, a group assignment $g$ in the group space $G$, and a label $y$ in the label space $Y$.
    % In the binary classification case, the label space $Y = \{0, 1\}$, while in regression case, the label space $Y = \mathbb{R}$.
    \item Binary Classification:
        \begin{itemize}
            \item The label space $Y$ is $\{-1, +1\}$.
            \item We want to find a fair classifier $\eta \in \mathbb{R}^k \rightarrow [0, 1]$ such that it takes a feature $\x \in \R^k$ and outputs a score $v = \eta(\x)$ that models the conditional probability $\Pr(y_i = +1 | \x)$, i.e. the confidence score of an instance with feature vector $\x$ to have a positive label $y = +1$, while preserving two out of the three fair conditions presented in \cite{kleinberg2016inherent} as follows.
            \item Condition (A) Calibration within the group: for each group $t$ and each score $v$, the probability of a instance in group $t$ with score $v$ which has label $1$ should be equal to $v$ fraction of the probability of a instance in group $t$ with score $v$. In other words, $ \forall t \in G, v \in [0, 1],$
            \begin{align*}
                \Pr_{(\x, g, y) \in Z}(g = t, \eta(\x) = v, y = +1) \\
                = v \cdot \Pr_{(\x, g, y) \in Z}(g = t, \eta(\x) = v)
            \end{align*}
            \item Condition (B) Balance for negative class: the average score of the instances with label $-1$ should be the same across groups. That is, $\forall s, t \in G,$
            \begin{align*}
                \int_{0}^{1} v \Pr_{(\x, g, y) \in Z}(\eta(\x) = v| g = s, y = -1) dv \\
                = \int_{0}^{1} v \Pr_{(\x, g, y) \in Z}(\eta(\x) = v| g = t, y = -1) dv
            \end{align*}
            \item Condition (C) Balance for positive class: the average score of the instances with label $+1$ should be the same across groups. That is, $\forall s, t \in G,$
            \begin{align*}
                \int_{0}^{1} v \Pr_{(\x, g, y) \in Z}(\eta(\x) = v| g = s, y = +1) dv \\
                = \int_{0}^{1} v \Pr_{(\x, g, y) \in Z}(\eta(\x) = v| g = t, y = +1) dv
            \end{align*}
        \end{itemize}
    \item 
  \end{itemize}
